import os
import streamlit as st
from dotenv import load_dotenv
from google import genai

# Bizana variables ziri muri .env kuri local machine
load_dotenv()

# Banza ushake muri Streamlit Secrets, niba ntabyo ushake muri Environment Variables (.env)
api_key = st.secrets.get("GEMINI_API_KEY") or os.getenv("GEMINI_API_KEY")

if not api_key:
    st.error("API Key ntabwo yabonetse! Yishyire muri .env cyangwa muri Streamlit Secrets.")
    st.stop()

client = genai.Client(api_key=api_key)
